// Updated bounding box for Lothian and Henry Islands with expanded northern boundary
var geometry = ee.Geometry.Rectangle([88.26, 21.53, 88.40, 21.72]);

Map.centerObject(geometry, 12);
Map.addLayer(geometry, {color: 'blue'}, 'Expanded Lothian & Henry Islands Box');


// Year to analyze: change this to desired year 2015-2022
var year = 2018;

// Load Landsat 8 Collection 2 Level 2, filtered by date and location
var landsat = ee.ImageCollection('LANDSAT/LC08/C02/T1_L2')
  .filterDate('2018-01-01','2018-12-31')
  .filterBounds(geometry)
  // Cloud mask using QA_PIXEL band
  .map(function(image) {
    var qa = image.select('QA_PIXEL');
    var cloudShadowBitMask = 1 << 3;
    var cloudsBitMask = 1 << 5;
    var mask = qa.bitwiseAnd(cloudShadowBitMask).eq(0)
      .and(qa.bitwiseAnd(cloudsBitMask).eq(0));
    return image.updateMask(mask);
  })
  .median()
  .clip(geometry);

// Import your training points FeatureCollections here,
// each with property 'landcover' = 1 (WaterBodies, blue), 2 (BuiltUp, red),
// 3 (Vegetation, green), 4 (BareLand, yellow)
// Example:
// var WaterBodies = ee.FeatureCollection('users/your_username/WaterBodies');
// var BuiltUp = ee.FeatureCollection('users/your_username/BuiltUp');
// var Vegetation = ee.FeatureCollection('users/your_username/Vegetation');
// var BareLand = ee.FeatureCollection('users/your_username/BareLand');
//
// Uncomment and edit above lines accordingly

// Combine training points for sampling
var trainingPoints = WaterBodies.merge(BuiltUp).merge(Vegetation).merge(BareLand);

// Select spectral bands for classification
var bands = ['SR_B2','SR_B3','SR_B4','SR_B5','SR_B6','SR_B7'];

// Sample input imagery at training points
var training = landsat.select(bands).sampleRegions({
  collection: trainingPoints,
  properties: ['landcover'],
  scale: 30
});

// Train Random Forest classifier with 50 trees
var classifier = ee.Classifier.smileRandomForest(50)
  .train(training, 'landcover', bands);

// Classify the composite image
var classified = landsat.select(bands).classify(classifier);

// Visualization palette for classes in order: WaterBodies, BuiltUp, Vegetation, BareLand
var palette = ['0000FF', 'FF0000', '00FF00', 'FFFF00'];

Map.addLayer(landsat, {bands:['SR_B5','SR_B4','SR_B3'], min:0, max:3000}, year + ' Landsat Composite');
Map.addLayer(classified, {min:1, max:4, palette: palette}, 'LULC Classification ' + year);

// Accuracy assessment on training data
var confusionMatrix = classifier.confusionMatrix();
print('Confusion Matrix:', confusionMatrix);
print('Overall Accuracy:', confusionMatrix.accuracy());
print('Kappa Coefficient:', confusionMatrix.kappa());
